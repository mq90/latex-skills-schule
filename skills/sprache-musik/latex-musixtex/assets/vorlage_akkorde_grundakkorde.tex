% \Quintenzirkel{Dur-Namen}{Moll-Namen}: kleines Rad, außen Dur, innen Moll
\newcommand{\Quintenzirkel}[2]{%
  \begin{tikzpicture}[font=\sffamily]
    \foreach \i in {0,...,11}{
      \fill[qz\i] (0,0) -- ({105-30*\i}:2.05) arc ({105-30*\i}:{75-30*\i}:2.05) -- cycle;
      \draw[white, line width=0.8pt] (0,0) -- ({105-30*\i}:2.05); }
    \fill[white] (0,0) circle (0.55);
    \draw[white, line width=0.8pt] (0,0) circle (1.3);
    \foreach \n [count=\i from 0] in {#1}{ \node[font=\sffamily\bfseries\small] at ({90-30*\i}:1.68) {\n}; }
    \foreach \n [count=\i from 0] in {#2}{ \node[font=\sffamily\scriptsize] at ({90-30*\i}:0.93) {\n}; }
  \end{tikzpicture}}

% \Grundakkord{Zeile}{Farbe 0–11}{Name}{Zusatz}{Vorzeichen}{Stufe}{Töne Klavier}{Tonnamen}
\newcommand{\Grundakkord}[8]{%
  \begin{scope}[shift={(0,{-1.83*#1})}]
    \fill[qz#2, rounded corners=1.5pt] (0,-0.15) rectangle (0.32,1.35);
    \node[font=\sffamily\bfseries\Large, anchor=west] at (0.55,0.72) {#3};
    \node[font=\sffamily\scriptsize, text=black!60, anchor=west] at (0.57,0.22) {#4};
    \begin{scope}[shift={(3.1,0.18)}, x=1pt, y=1pt]
      \Notenlinien{118}
      \Taktstrich{118}
      \Vorzeichnung{31}{#5}
      \Dreiklang{92}{#6}{0}{0}{0}
    \end{scope}
    \Klaviatur[shift={(8.1,-0.05)}, x=3.7mm, y=13mm]{#7}
    \node[font=\sffamily\large, anchor=west, align=left] at (13.8,0.6) {#8};
    \draw[black!15, line width=0.4pt] (0,-0.42) -- (18.4,-0.42);
  \end{scope}}

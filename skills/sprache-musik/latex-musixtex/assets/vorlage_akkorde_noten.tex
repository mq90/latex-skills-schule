% Eintrag: Trenner/Name/Stufe/V1/V2/V3/Tonnamen
%   Trenner: 0 = Zeilenanfang, 1 = Taktstrich, 2 = gestrichelt mit „=“ (enharmonisch gleich)
%   Stufe: 0 = e' (unterste Linie), -2 = c'; V1–V3: 0, 52 (Kreuz), 50 (Be), 53 (Doppelkreuz), 51 (Doppel-Be)
\newcommand{\Akkordreihe}[2]{%
  \begin{tikzpicture}[x=1pt, y=1pt, baseline=0pt]
    \Notenlinien{520}
    \Taktstrich{520}
    \pgfmathsetmacro{\feld}{(520-36)/#1}
    \foreach \trenn/\name/\stufe/\va/\vb/\vc/\toene [count=\i from 0] in {#2}{
      \pgfmathsetmacro{\xl}{36+\feld*\i}
      \pgfmathsetmacro{\xm}{36+\feld*(\i+0.58)}
      \ifnum\trenn=1 \Taktstrich{\xl}\fi
      \ifnum\trenn=2 \Taktstrich[dash pattern=on 2.5pt off 2pt]{\xl}
        \node[font=\sffamily\bfseries\large, text=akzent] at (\xl,45) {=};\fi
      \node[font=\sffamily\bfseries, text=akzent, anchor=base] at (\xm,41) {\name};
      \Dreiklang{\xm}{\stufe}{\va}{\vb}{\vc}
      \node[font=\sffamily\scriptsize, text=black!60, anchor=base] at (\xm,-25) {\toene};
    }
  \end{tikzpicture}\par\vspace{9mm}}

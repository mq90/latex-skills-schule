% Eine Zeile: \Akkordzeile{Zeilennummer}{Name links}{Symbolstamm}{Grundton 0–11}
\newcommand{\Akkordzeile}[4]{%
  \node[font=\sffamily\bfseries\large, anchor=east, align=right] at (-0.25,{-1.75*#1+0.55}) {#2};
  \foreach \zusatz/\toene [count=\sp from 0] in {
      {}/{#4,#4+4,#4+7}, m/{#4,#4+3,#4+7}, 7/{#4,#4+4,#4+7,#4+10},
      maj7/{#4,#4+4,#4+7,#4+11}, m7/{#4,#4+3,#4+7,#4+10}}{
    \node[font=\sffamily\small, anchor=base west, inner sep=0] at ({3.42*\sp},{-1.75*#1+1.22}) {#3\zusatz};
    \Klaviatur[shift={({3.42*\sp cm},{-1.75*#1 cm})}, x=2.3mm, y=11mm]{\toene}
  }
}
